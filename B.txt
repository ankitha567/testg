import numpy as np, seaborn as sns, matplotlib.pyplot as plt
from sklearn.datasets import fetch_california_housing

# Load dataset
df = fetch_california_housing(as_frame=True).frame
num_features = df.select_dtypes('number').columns

# Histograms
plt.figure(figsize=(15,10))
for i,f in enumerate(num_features,1):
    plt.subplot(3,3,i)
    sns.histplot(df[f], kde=True, bins=30, color='blue')
    plt.title(f'Distribution of {f}')
plt.tight_layout(); plt.show()

# Boxplots
plt.figure(figsize=(15,10))
for i,f in enumerate(num_features,1):
    plt.subplot(3,3,i)
    sns.boxplot(x=df[f], color='orange')
    plt.title(f'Box Plot of {f}')
plt.tight_layout(); plt.show()

# Outlier detection (IQR)
print("Outliers Detection:")
for f in num_features:
    Q1,Q3 = df[f].quantile([.25,.75]); 
    IQR=Q3-Q1
    lb,ub = Q1-1.5*IQR, Q3+1.5*IQR
    print(f"{f}: {((df[f]<lb)|(df[f]>ub)).sum()} outliers")

# Summary
print("\nDataset Summary:\n", df.describe())

import numpy as np, matplotlib.pyplot as plt

def gaussian_kernel(x, xi, tau): 
    return np.exp(-np.sum((x - xi)**2)/(2*tau**2))

def lwr(x, X, y, tau):
    W = np.diag([gaussian_kernel(x, X[i], tau) for i in range(len(X))])
    theta = np.linalg.inv(X.T @ W @ X) @ (X.T @ W @ y)
    return x @ theta

# Data
np.random.seed(42)
X = np.linspace(0, 2*np.pi, 100); y = np.sin(X)+0.1*np.random.randn(100)
Xb = np.c_[np.ones(len(X)), X]

# Predictions
xt = np.linspace(0, 2*np.pi, 200); xtb = np.c_[np.ones(len(xt)), xt]
tau = 0.5
yp = np.array([lwr(xi, Xb, y, tau) for xi in xtb])

# Plot
plt.scatter(X, y, c='r', alpha=0.7, label='Training Data')
plt.plot(xt, yp, 'b', lw=2, label=f'LWR Fit (tau={tau})')
plt.xlabel('X'); plt.ylabel('y'); plt.title('Locally Weighted Regression')
plt.legend(); plt.grid(alpha=0.3); plt.show()

import numpy as np, matplotlib.pyplot as plt
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier, plot_tree
from sklearn.metrics import accuracy_score

# Load data
data = load_breast_cancer()
X_train, X_test, y_train, y_test = train_test_split(data.data, data.target, test_size=0.2, random_state=42)

# Train model
clf = DecisionTreeClassifier(random_state=42).fit(X_train, y_train)
y_pred = clf.predict(X_test)

# Accuracy
print(f"Model Accuracy: {accuracy_score(y_test, y_pred)*100:.2f}%")

# Predict new sample
pred = clf.predict([X_test[0]])[0]
print("Predicted Class for the new sample:", "Benign" if pred==1 else "Malignant")

# Plot tree
plt.figure(figsize=(12,8))
plot_tree(clf, filled=True, feature_names=data.feature_names, class_names=data.target_names)
plt.title("Decision Tree - Breast Cancer Dataset")
plt.show()

import matplotlib.pyplot as plt
from sklearn.datasets import fetch_olivetti_faces
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB

# Load data
data = fetch_olivetti_faces(shuffle=True, random_state=42)
X_train, X_test, y_train, y_test = train_test_split(data.data, data.target, test_size=0.3, random_state=42)

# Train & predict
model = GaussianNB().fit(X_train, y_train)
y_pred = model.predict(X_test)

print("Accuracy:", (y_pred == y_test).mean()*100)

# Show 4 predictions
fig, axes = plt.subplots(1, 4, figsize=(8, 3))
for ax, img, true, pred in zip(axes, X_test[:4], y_test[:4], y_pred[:4]):
    ax.imshow(img.reshape(64,64), cmap="gray")
    ax.set_title(f"T:{true},P:{pred}")
    ax.axis("off")
plt.show()

import matplotlib.pyplot as plt, seaborn as sns
from sklearn.cluster import KMeans
from sklearn.datasets import load_breast_cancer
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import classification_report

# Load & scale data
data = load_breast_cancer()
X = StandardScaler().fit_transform(data.data)

# K-Means clustering
kmeans = KMeans(n_clusters=2, random_state=42).fit(X)
y_kmeans = kmeans.labels_
print("Classification Report:\n", classification_report(data.target, y_kmeans))

# PCA for 2D visualization
X_pca = PCA(n_components=2).fit_transform(X)
centroids = PCA(n_components=2).fit(X).transform(kmeans.cluster_centers_)

# Plot clusters vs true labels
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12,5))
sns.scatterplot(x=X_pca[:,0], y=X_pca[:,1], hue=y_kmeans, palette="Set1", ax=ax1)
ax1.scatter(centroids[:,0], centroids[:,1], c="black", s=200, marker="X")
ax1.set_title("K-Means Clusters")

sns.scatterplot(x=X_pca[:,0], y=X_pca[:,1], hue=data.target, palette="coolwarm", ax=ax2)
ax2.set_title("True Labels")
plt.show()
