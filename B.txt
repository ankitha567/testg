#include <stdio.h>
#include <math.h>
#include <omp.h>

int is_prime(int n) {
    if (n < 2) return 0;
    for (int i = 2; i <= sqrt(n); i++)
        if (n % i == 0) return 0;
    return 1;
}

int main() {
    int n, c = 0;
    printf("Enter the value of n: ");
    scanf("%d", &n);

    printf("\nPrime numbers from 1 to %d:\n", n);
    for (int i = 1; i <= n; i++)
        if (is_prime(i)) printf("%d ", i);

    double t = omp_get_wtime();
    for (int i = 1; i <= n; i++) c += is_prime(i);
    printf("\nSerial Time: %f seconds\n", omp_get_wtime() - t);

    t = omp_get_wtime();
    #pragma omp parallel for reduction(+:c)
    for (int i = 1; i <= n; i++) c += is_prime(i);
    printf("Parallel Time: %f seconds\n", omp_get_wtime() - t);
    return 0;
}

#include <mpi.h>
#include <stdio.h>

#define DEADLOCK_PART 2   // 1 = deadlock, 2 = deadlock avoidance

int main(int argc, char *argv[]) {
    int rank, size, num = 123;
    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (size < 2) {
        if (rank == 0) printf("Run with at least 2 processes.\n");
        MPI_Finalize();
        return 0;
    }

    int other = 1 - rank;   // partner process (0 <-> 1)

#if DEADLOCK_PART == 1
    // Both processes receive first -> deadlock
    if (rank < 2) {
        printf("Process %d waiting to receive from Process %d...\n", rank, other);
        MPI_Recv(&num, 1, MPI_INT, other, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
        MPI_Send(&num, 1, MPI_INT, other, 0, MPI_COMM_WORLD);
    }
#else
    // Process 0 sends first, Process 1 receives first -> no deadlock
    if (rank == 0) {
        MPI_Send(&num, 1, MPI_INT, 1, 0, MPI_COMM_WORLD);
        MPI_Recv(&num, 1, MPI_INT, 1, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
        printf("Process 0 received back number: %d\n", num);
    } else if (rank == 1) {
        MPI_Recv(&num, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
        MPI_Send(&num, 1, MPI_INT, 0, 0, MPI_COMM_WORLD);
        printf("Process 1 received and sent back number: %d\n", num);
    }
#endif

    MPI_Finalize();
    return 0;
}
